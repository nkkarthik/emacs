* Systemd user unit

#+name: l.service
#+begin_src conf :tangle ~/.config/systemd/user/l.service
[Unit]
Description=Jupyter Lab

[Service]
Type=simple
WorkingDirectory=%h/e/a
ExecStart=%h/l/venv/bin/jupyter lab --dev-mode --notebook-dir=%h --ServerApp.preferred_dir=%h/e/a --ServerApp.ip=0.0.0.0 --ServerApp.allow_remote_access=true --ServerApp.token=moonsoon
Environment=PATH=%h/bin:%h/l/venv/bin:%h/.local/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin
Restart=always
RestartSec=10

[Install]
WantedBy=default.target

#+end_src
